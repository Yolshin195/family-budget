[extra.faq]
title = "Questions and answers"
[[extra.faq.items]]
q = "Where is my data stored and who can see it?"
a = "On our server, with encrypted backups. Only a project's members see its data. Passwords are kept by the Rauthy sign-in server — the app never sees them."
[[extra.faq.items]]
q = "How do I invite my family?"
a = "Create a family budget or turn your personal one into a family budget, then invite members with a link in a messenger, a QR code or by email. Roles: owner, editor, viewer. A link is valid for 7 days and works for one person."
[[extra.faq.items]]
q = "How do I connect Claude, and what can it do with my data?"
a = "In Profile → Security → Assistants choose a project and permissions, then add the connector in Claude. It can view the budget, add and edit transactions, manage scheduled operations, directories (currencies, accounts, categories, rates) and limits — only what you ticked. Nothing can be deleted through Claude."
[[extra.faq.items]]
q = "Which currencies are supported?"
a = "Any ISO 4217 code: baht, ruble, dollar, euro and more. Each account keeps its own currency, and an exchange between accounts is one transaction with the fee and the real rate. The budget converts to your base currency at the rate on the transaction date."
[[extra.faq.items]]
q = "Can I use it on my phone?"
a = "Yes: the main sections and a round + button for expenses sit at the bottom of the screen. Add the site to your home screen and it opens like an app. And with Claude on your phone you can dictate expenses."
[[extra.faq.items]]
q = "Can I export my data?"
a = "Yes: create a personal API token in Profile → Security and fetch transactions, accounts and directories via the REST API. A token works in one project, with the permissions and expiry you choose."
[[extra.faq.items]]
q = "How do I delete my account?"
a = "Press “Delete account” in your profile. Personal data is deleted in the app and on the sign-in server. See the Data deletion page for details."
